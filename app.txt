from flask import Flask, render_template, request
import joblib
import numpy as np

app = Flask(__name__)

# Load vectorizer and model only once
vectorizer = joblib.load("tfidf_vectorizer.pkl")
model = joblib.load("baseline_model.pkl")

# Sample demo emails
demo_emails = [
    "Congratulations! You have won a $1000 gift card.",
    "Please find attached the meeting agenda for tomorrow.",
    "Urgent: Your account has been compromised. Reset your password now.",
    "Hey, just wanted to check if you're free for lunch.",
    "Limited time offer! Get 70% off on all products today."
]

@app.route('/', methods=['GET', 'POST'])
def home():
    user_email = ""
    user_prediction = None

    if request.method == 'POST':
        user_email = request.form.get('email_text', '').strip()
        if user_email:
            X = vectorizer.transform([user_email]).toarray()
            user_prediction = model.predict(X)[0]

    # Predict demo emails
    demo_preds = model.predict(vectorizer.transform(demo_emails).toarray())
    results = list(zip(demo_emails, demo_preds))

    return render_template(
        'index.html',
        results=results,
        user_email=user_email,
        user_prediction=user_prediction
    )

if __name__ == "__main__":
    app.run(debug=True)
